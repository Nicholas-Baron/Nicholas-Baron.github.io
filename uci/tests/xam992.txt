main
var x, y, z;

function adder(a, b);
{
    return b + a;
};

{
    let x <- call InputNum;
    let y <- x;
    let z <- 0;

    while z < 10 do
        let y <- call adder(x, z);
        while y < 10 do 
            let y <- call adder(y, y);
            if z < 5 then
                let z <- y;
            fi;
        od;
        let z <- z * 2;
    od;
}.
